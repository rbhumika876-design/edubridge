// progress.js

// 1. This function saves a video or test into the browser's memory
function saveProgress(videoId) {
    // Get the current list of completed items from memory, or start an empty list
    let completedItems = JSON.parse(localStorage.getItem('userProgress')) || [];
    
    // If this item isn't in the list yet, add it
    if (!completedItems.includes(videoId)) {
        completedItems.push(videoId);
        // Save the updated list back to the browser's memory
        localStorage.setItem('userProgress', JSON.stringify(completedItems));
    }
}

// 2. This function calculates the percentage and visually updates the progress bar
function drawProgressBar() {
    let completedItems = JSON.parse(localStorage.getItem('userProgress')) || [];
    
    // Change this number to match the total number of items you have
    let totalItems = 4; 
    let currentCompleted = completedItems.length;
    
    // Calculate percentage
    let percentage = Math.round((currentCompleted / totalItems) * 100);
    if (percentage > 100) percentage = 100; // Keep it maxed at 100%

    // Find the elements on the screen and update them
    let progressBar = document.getElementById('my-bar');
    let progressText = document.getElementById('my-text');

    if (progressBar && progressText) {
        progressBar.style.width = percentage + '%';
        progressText.innerText = percentage + '% Completed';
    }
}

// 3. Make sure the progress bar updates automatically whenever a page loads
window.addEventListener('DOMContentLoaded', drawProgressBar);
